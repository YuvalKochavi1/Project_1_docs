\section{Chapter 1 - Definitions in Radiative Transfer}
\subsection{The radiation Field}
\begin{itemize}
    \item The definition of the PDF for photons in phase space uses the fact that $E=h\nu$ in order to parameterize the phase space by $(\bm{r},\nu,\bm{\Omega})$, i.e. six DOF.
    \begin{align}
        f(t)=f(\bm{r},\nu,\bm{\Omega}, t)\\
        dn=fd\bm{r}d\nu d\bm{\Omega}
    \end{align}
    *$d\bm{r}$ is actually a volume differential.
    \item The specific intensity and radiant energy are:
    \begin{align}
        I(\bm{r},\nu,\bm{\Omega}, t)=ch\nu f(\bm{r},\nu,\bm{\Omega}, t)\\
        dE=I(\bm{r},\nu,\bm{\Omega}, t)\cos(\theta)d\nu d\bm\Omega d\sigma dt
    \end{align}
    *Notice that $Id\bm\Omega d\nu dt$ is the number of photons in the cone seen in the following figure, and multiplying it by $\cos\theta d\sigma$ gives $dE$.
\end{itemize}
\subsection{The energy flux, radiative flux and pressure tensor}
\begin{itemize}
    \item The energy density of photons comes from simply integrating the energy of all photons in intervals $d\nu,d\Omega$ (in phase space):
    \begin{align}
         u(\bm r,t) = \int_0^\infty d\nu\int_{4\pi} d\bm\Omega h\nu f=\frac{1}{c}\int_0^\infty d\nu \int_{4\pi}d\bm\Omega I
    \end{align}
    
    \underline{Example:} Given the specific intensity of Planck function (black body), we can calculate $u(\bm r,t)$ and show that $u=aT^4$.
    \item If a surface element is oriented perpendicular to the x axis, then the amount of $dE$ passing through it per unit time and surface is (analogous to the calculation of I and dE from beforehand, but now $d\Omega_x=d\sigma\cos\theta$):
\begin{align}
        F_x (\bm r,t)= \int_0^\infty d\nu\int_{4\pi}d\bm\Omega ch\nu\Omega_xf=\int_0^\infty d\nu \int_{4\pi} d\bm\Omega \Omega_x I
\end{align}
and in general we define:
\begin{align}
    \bm F=\int_0^\infty d\nu \int_{4\pi} \bm d\Omega \bm \Omega I
\end{align}
If $I$ is isotropic then $F=0$ (in thermodynamic equillibrium there shouldn't be net energy flow).
\item The rate of momentum flow across a surface is the momentum tensor. For example, $p_{xy}(\bm r,t)$ is the rate of y momentum flow per unit area \& time across a surface perp. to x axis. A photon caries $y$ momentum in the amount $\frac{h\nu\Omega_y}{c}$ so:
\begin{align}
    p_{xy}=\int_0^\infty d\nu \int_{4\pi} d\bm\Omega (c\Omega_x)\left(\frac{h\nu\Omega_y}{c}\right) f=\frac{1}{c}\int_0^\infty d\nu \int_{4\pi} d\bm\Omega \Omega_x\Omega_y I
\end{align}
and in general:
\begin{align}
    \bm p(\bm r,t)==\frac{1}{c}\int_0^\infty d\nu \int_{4\pi} d\bm\Omega \bm\Omega \bm\Omega I
\end{align}
\underline{Note:} The pressure is symmetric $p_{kj}=p_{jk}$ and since $\Omega_x^2+\Omega_y^2+\Omega_z^2=1$ we have:
\begin{align}
    p_{xx}+p_{yy}+p_{zz} = \frac{1}{c}\int_0^\infty d\nu \int_{4\pi} d\bm\Omega I=u
\end{align}
This leads to the definition of the average pressure:
\begin{align}
    \bar{p}=\frac{1}{3}(p_{xx}+p_{yy}+p_{zz})=\frac{1}{3}u
\end{align}
\underline{Examples:} For an isotropic specific intensity one finds that
\begin{align}
    p_{xx}=p_{yy}=p_{zz}=\bar{p}=\frac{1}{3}u
\end{align}
with all off diagonal terms of the tensor zero.
On the other extreme from isotropy, if at $(\bm r,t)$ all photons are moving on the $x$ direction (the specific intensity is the Dirac Delta function in the x direction) then $p_{xx}=u$ with all other 8 components zero.
\end{itemize}
\subsection{Interaction of the Radiation Field with Matter}
\begin{itemize}
    \item The probability density (of $(\bm r,\nu)$) of a photon to be absorbed travelling a distance $ds$ is $\sigma_a(\bm r,\nu,t)ds$. 
\item \underline{Note:} The independency of $\bm\Omega$ is caused by the lack of prefered direction in the medium (the material).
\item The probability density (of $\bm r$) of a photon to be scattered from $\nu'$ to $\nu$ (centered at $d\nu$) and from $\Omega'$ to $\Omega$ centered at $d\Omega$, travelling distance $ds$ is:
\begin{align}
    probability = \sigma_s(\nu'\to\nu,\Omega'\cdot\Omega)d\nu d\bm\Omega ds
\end{align}
\underline{Note:} The notation $\Omega'\cdot\Omega$ implies the lack of preferred direction.
\item Integrating over $d\nu,d\bm\Omega$ gives the probability density (of $\bm r,t,\nu'$) for a general event of scattering to happen:
\begin{align}
    \sigma_s(\nu')=\int_0^\infty d\nu \int_{4\pi} d\bm\Omega \sigma_s(\nu'\to \nu,\Omega'\cdot\Omega)=2\pi\int_0^\infty d\nu \int_{-1}^{1}d\mu_{0}\sigma_s(\nu'\to\nu,\mu_0)
\end{align} 
where here $\Omega'$ is just an arbitrary angle chosen to be along the z direction (and defining $\mu_0=\Omega'\cdot\Omega$, or we can think of the third term as the original intention, which is the integration of the $\mu_0=\cos\theta$ term that corresponds to the scattering angle (regardless of the two angles themselves).
\item One can define
\begin{align}
    K(\nu'\to\nu,\Omega\cdot\Omega')=\frac{\sigma_s(\nu'\to\nu,\Omega'\cdot\Omega)}{\sigma_s(\nu')}
\end{align}
as the probability of scattering to $(\nu,\Omega)$ given that a scattering event has happened at $(\nu',\Omega')$ .
\item The scattering is called \textit{coherent} if there is no change in $\nu$ (i.e. energy) on scattering events, and is called \textit{isotropic} if the kernel is independent of the scattering angle (i.e. once a scattering event has occurred, the scattered angle is equally distributed).
\item $\tilde{\omega}=\frac{\sigma_s}{\sigma_s+\sigma_a}$ denotes the probability of scattering once known that an event has occurred.
\item  The number of collided photons a beam in an interval of distance $ds$ satisfies:
\begin{align}
    dN=-N\sigma ds \\
    \frac{dN}{ds}=-N\sigma \\
    N=N_0e^{-as}
\end{align}
Therefore, we can think of probability to event occurrence after travelling a distance $s$, inside a further interval of length $ds$ and it is 
\begin{align}
    \bar s=\lambda=\frac{\int_0^s ds\left(N_0\sigma e^{-as}\right)}{\int_0^\infty ds\left(N_0\sigma e^{-as}\right)}
\end{align}
\item \underline{Important Note:} We notice that the probability of a single photon to be scattered or absorbed after traveling distance $s$ is Geo($\sigma$) and the mean distance it passes is $\lambda=\frac{1}{\sigma}$.
\item \underline{Emission:} $\text{photons emitted per unit volume} = q(\bm r,\nu,t)d\nu d\bm\Omega$ and it's independent of $\Omega$ (material has no preferred direction to which it spontaneously emits photons).

\end{itemize}
\section{Chapter 2 - The Equation of Transfer}
\subsection{Eulerian Derivation}
\begin{itemize}
    \item change in photons in a cube in phase space = $\Delta V \frac{\partial}{\partial t}
    f(\bm r,\nu,\bm\Omega)$ where $\Delta V=\Delta x \Delta y \Delta z \Delta \mu\Delta\varphi\Delta \nu$
These are the 5 processes which lead to change in $f$:
\begin{enumerate}
    \item Stream - Streaming of photons in or out of the cube.
    \item Absorption - material spontaneously absorbs photons.
    \item Outscattering - scattering from $\nu,\Omega$ to all other freq. and directions.
    \item Inscattering - scattering into $\nu,\Omega$
    \item Emission - material spontaneously creates photons.

\end{enumerate}
\item \underline{Stream} The rate of photons leaving the cube through the surfaces perp. to the x axis is:
\begin{align}
    (streaming)_x=c\Omega_xf(\bm r,\nu,\Omega,t)|^{x+\Delta x}_x\Delta y\Delta z\Delta \nu\Delta \mu\Delta \varphi \\
    (streaming_x)=\Delta V \frac{\partial}{\partial x}\left[\dot x f(\bm r,\nu,\bm\Omega,t)\right]
\end{align}
when the second equation comes from defining $\dot x=c\Omega_x$ and then multiplying and dividing by $\Delta x\to dx\to0$.
The total streaming then becomes:
\begin{align}
    streaming = \left[\frac{\partial}{\partial x}(\dot xf)+\frac{\partial}{\partial y}(\dot yf)+...\right] \Delta V
\end{align}
\underline{Note:} Terms like $\dot \nu,\dot \varphi$ denote the rate of change of the $\nu,\varphi$ values of the photon as it streams along its path.
\item\underline{Other Contributions:} Appear clearly in the book. Notice that: \begin{enumerate}
    \item Scattering and absorption is calculated through = probability for a single photon to be scattered/absorbed \underline{per unit time} (hence use of $c$ instead of $ds$) $\times$ number of photons in the cube $f\Delta V$
    \item In scattering, the in and out both depend on two other integration parameters $\nu',\Omega'$. In outscattering the $f$ can be taken out of the integral and we remain with the total differential scattering coefficient $\sigma_s(\bm r,\nu'\to\nu,\Omega'\cdot\Omega,t)$ which corresponds to the total photons reaching $(\nu,\Omega)$ per unit time.
    \item substituting $I$ and/or $S=h\nu q(\bm r,\nu,t)$ (Energy emitted per unit time and volume) lead to different forms of the equation.
    \item In case of photons $\dot \varphi=0, \dot \nu=0$ (angle and frequency of a photon don't change between collisions. Therefore, one can manage a simpler version of the equation. 

\end{enumerate}

\end{itemize}
