\title{A Unified Gray Hammer--Rosen Model\\for Surface- and Bath-Temperature Drives}
\author{Menahem Krief}
\date{\today}

\begin{document}
\maketitle

\begin{abstract}
A compact Hammer--Rosen-type model is developed for planar gray,
two-temperature Marshak waves.  The model allows independent Rosseland and
absorption opacity power laws and can be driven either by a prescribed surface
radiation temperature $\Ts(t)$ or by a prescribed bath temperature $\Tb(t)$
through a Marshak boundary condition.  The radiation profile uses the
first-order Hammer--Rosen curvature correction.  The material profile is the
same radiation shape multiplied by a monotone factor $\exp[-s(t)y]$, where the
single scalar $s(t)$ is obtained from an areal material-energy closure.  The
surface-driven model requires two scalar physical ODEs, three running time
quadratures, and one scalar root.  The bath-driven model adds only one scalar
ODE for $\Ts(t)$.  No spatial mesh, diffusion matrix, PDE solve, or
history-dependent integral equation is required.  A fully self-contained
implementation recipe is given in the final section.
\end{abstract}

\section{Gray diffusion problem}

Consider a planar, fixed-density material occupying $x>0$.  Hydrodynamic
motion is neglected.  The radiation energy density and material energy density
are
\begin{equation}
 E_r=\arad\Tr^4,
 \qquad
 U=f\rho^{1-\mu}T^\beta,
 \qquad
 \arad=\frac{4\sigma_{\rm SB}}{c}.
 \label{eq:energies}
\end{equation}
The Rosseland and absorption opacities are independent power laws,
\begin{equation}
 \frac{1}{\kappa_R}=gT^\alpha\rho^{-\lambda},
 \qquad
 \frac{1}{\kappa_a}=g'T^{\alpha'}\rho^{-\lambda'}.
 \label{eq:opacities}
\end{equation}
Thus $(\alpha,\lambda,g)$ are the Rosseland parameters and
$(\alpha',\lambda',g')$ are the absorption parameters.

The underlying gray diffusion equations are
\begin{align}
 \frac{\partial}{\partial t}\left(\arad\Tr^4\right)
 &=\frac{\partial}{\partial x}
 \left[
 \frac{cg}{3}\rho^{-1-\lambda}T^\alpha
 \frac{\partial}{\partial x}\left(\arad\Tr^4\right)
 \right]
 -\frac{\arad c}{g'}\rho^{1+\lambda'}T^{-\alpha'}
 \left(\Tr^4-T^4\right),
 \label{eq:gray_radiation}\\
 \frac{\partial}{\partial t}
 \left(f\rho^{1-\mu}T^\beta\right)
 &=\frac{\arad c}{g'}\rho^{1+\lambda'}T^{-\alpha'}
 \left(\Tr^4-T^4\right).
 \label{eq:gray_material}
\end{align}
The reduced model derived below does not solve
\eqref{eq:gray_radiation}--\eqref{eq:gray_material} pointwise.

Two boundary specifications are considered.

\paragraph{Surface-temperature drive.}
\begin{equation}
 \boxed{\Tr(0,t)=\Ts(t),}
 \label{eq:Ts_bc}
\end{equation}
where $\Ts(t)$ is prescribed.

\paragraph{Bath-temperature drive.}
The surface radiation temperature is unknown and the incoming bath is imposed
through the Marshak condition
\begin{equation}
 \boxed{
 F_s(t)=\frac{\arad c}{2}
 \left[\Tb^4(t)-\Ts^4(t)\right],
 \qquad \Ts(t)=\Tr(0,t),}
 \label{eq:marshak_bc}
\end{equation}
where $F_s>0$ denotes flux into the material.

Define the two standard Hammer--Rosen parameters
\begin{equation}
 \boxed{
 \eps=\frac{\beta}{4+\alpha},
 \qquad
 C=\frac{4\arad c g}{3(4+\alpha)f}
 \rho^{-2+\mu-\lambda}.}
 \label{eq:eps_C}
\end{equation}
The exponent of the leading front profile will always be written explicitly as
$1/(4+\alpha-\beta)$; no separate symbol is introduced for it.

\section{Surface material temperature}

Let
\begin{equation}
 \boxed{\Tm(t)=T(0,t)}
 \label{eq:Tm_def}
\end{equation}
be the surface material temperature.  Evaluating the local material balance at
$x=0$ gives
\begin{resultbox}
\begin{equation}
 \boxed{
 \frac{\dif}{\dif t}
 \left[f\rho^{1-\mu}\Tm^\beta(t)\right]
 =\frac{\arad c}{g'}\rho^{1+\lambda'}
 \Tm^{-\alpha'}(t)
 \left[\Ts^4(t)-\Tm^4(t)\right].}
 \label{eq:surface_material_ode}
\end{equation}
\end{resultbox}
Equivalently,
\begin{equation}
 \boxed{
 \dot\Tm(t)
 =\frac{\arad c}{\beta f g'}\rho^{\mu+\lambda'}
 \Tm^{1-\alpha'-\beta}(t)
 \left[\Ts^4(t)-\Tm^4(t)\right].}
 \label{eq:surface_material_ode_T}
\end{equation}
This equation naturally gives $\Tm<\Ts$ during heating and may give
$\Tm>\Ts$ during cooling.

Equation \eqref{eq:surface_material_ode} is one scalar, generally stiff ODE.
In Python it should be integrated with
\texttt{solve\_ivp}, using \texttt{method="Radau"} or
\texttt{method="BDF"}.  At a perfectly cold start, a code may internally
evolve the regular power $\Tm^{\alpha'+\beta}$, but that auxiliary variable is
only a numerical device and is not part of the analytical model.

\section{Common-shape gray Hammer--Rosen front}

\subsection{Gray storage coordinate}

For the leading common-shape reduction, write locally
\begin{equation}
 \Tr(x,t)=\Ts(t)q(x,t),
 \qquad
 T(x,t)=\Tm(t)q(x,t),
 \qquad 0\le q\le1.
 \label{eq:common_shape}
\end{equation}
The ratio of radiation energy to material energy at the surface is
\begin{equation}
 \boxed{
 R(t)=\frac{\arad\Ts^4(t)}
 {f\rho^{1-\mu}\Tm^\beta(t)}.}
 \label{eq:R}
\end{equation}
At an arbitrary profile value $q$, the corresponding ratio is
$R(t)q^{4-\beta}$.  The gray storage coordinate entering the HR reduction is
therefore
\begin{equation}
 Z(q;R)
 =(4+\alpha-\beta)
 \int_0^q
 \frac{\xi^{3+\alpha-\beta}}
 {1+R\xi^{4-\beta}}\dif\xi.
 \label{eq:Z_gray}
\end{equation}
The surface value $Z(1;R)$ is the gray storage factor
\begin{equation}
 \boxed{
 G(R)
 =(4+\alpha-\beta)
 \int_0^1
 \frac{\xi^{3+\alpha-\beta}}
 {1+R\xi^{4-\beta}}\dif\xi.}
 \label{eq:G_integral}
\end{equation}
The denominator is the normalized total storage
$1+R\xi^{4-\beta}$; thus $G$ reduces the effective drive whenever radiation
energy is non-negligible.

The integral has the exact Gauss-hypergeometric form
\begin{resultbox}
\begin{equation}
 \boxed{
 G(R)
 ={}_2F_1\!\left(
 1,
 \frac{4+\alpha-\beta}{4-\beta};
 1+\frac{4+\alpha-\beta}{4-\beta};
 -R
 \right).}
 \label{eq:G_hypergeometric}
\end{equation}
\end{resultbox}
The identity follows from
\begin{equation}
 {}_2F_1(a,b;c;z)
 =\frac{\Gamma(c)}{\Gamma(b)\Gamma(c-b)}
 \int_0^1
 u^{b-1}(1-u)^{c-b-1}(1-zu)^{-a}\dif u,
 \label{eq:2F1_integral}
\end{equation}
valid initially for $\Re c>\Re b>0$ and elsewhere by analytic continuation.
In Python, \eqref{eq:G_hypergeometric} is evaluated directly with
\texttt{hyp2f1}.

\subsection{Base front}

Define temporarily the running history
\begin{equation}
 \calI(t)=\int_0^t
 \Ts^4(t')\Tm^\alpha(t')G(R(t'))\dif t'.
 \label{eq:I_history}
\end{equation}
Applying the first-order HR front moment to the gray storage coordinate gives
\begin{resultbox}
\begin{equation}
 \boxed{
 \xfzero^2(t)
 =\frac{2+\eps}{1-\eps}
 C\Tm^{-\beta}(t)\calI(t).}
 \label{eq:xf0_I}
\end{equation}
\end{resultbox}
Equivalently, without introducing $\calI$,
\begin{equation}
 \boxed{
 \xfzero^2(t)
 =\frac{2+\eps}{1-\eps}
 C\Tm^{-\beta}(t)
 \int_0^t
 \Ts^4(t')\Tm^\alpha(t')G(R(t'))\dif t'.}
 \label{eq:xf0}
\end{equation}
Numerically this is one scalar running quadrature.  It is advanced together
with the ODE states; no past time array needs to be reintegrated at every step.

\section{First-order radiation profile and the two boundary drives}

The normalized first-order HR radiation profile is
\begin{equation}
 \boxed{
 \Tr(x,t)=\Ts(t)
 \left[
 \left(1-\frac{x}{\xf(t)}\right)
 \left(1+B(t)\frac{x}{\xf(t)}\right)
 \right]^{1/(4+\alpha-\beta)}.}
 \label{eq:radiation_profile}
\end{equation}
The coefficient $B(t)$ is the first HR curvature correction.  The profile
requires $B(t)>-1$ so that its base remains nonnegative on the wave interval.

\subsection{Prescribed surface radiation temperature}

For prescribed $\Ts(t)$, the HR correction is
\begin{resultbox}
\begin{equation}
 \boxed{
 \begin{aligned}
 B(t)=\frac{\eps}{2}\Bigg[1
 -\frac{2+\eps}{1-\eps}
 &\frac{\displaystyle
 \int_0^t\Ts^4(t')\Tm^\alpha(t')G(R(t'))\dif t'}
 {\Ts^4(t)\Tm^\alpha(t)G(R(t))}
 \\
 &\times\frac{\dif}{\dif t}\ln\Ts^{4+\alpha}(t)
 \Bigg].
 \end{aligned}}
 \label{eq:B_Ts}
\end{equation}
\end{resultbox}
This is an algebraic post-processing formula once the base-front history is
known.  For a tabulated drive, use a shape-preserving interpolant such as
\texttt{scipy.interpolate.PchipInterpolator}; its analytical derivative can
be obtained with \texttt{PchipInterpolator.derivative()}.  If the tabulated
pulse contains measurement noise, regularize only this derivative, not the
drive used in the energy quadratures.

\subsection{Prescribed bath temperature}

At the surface, the diffusion flux predicted by
\eqref{eq:radiation_profile}, using the base front inside the boundary closure,
is
\begin{equation}
 F_s(t)
 =\frac{4\arad c g}{3(4+\alpha-\beta)\xfzero(t)}
 \rho^{-1-\lambda}\Tm^\alpha(t)\Ts^4(t)[1-B(t)].
 \label{eq:surface_flux_profile}
\end{equation}
Equating this to the Marshak flux \eqref{eq:marshak_bc} gives
\begin{resultbox}
\begin{equation}
 \boxed{
 B(t)=1-
 \frac{3(4+\alpha-\beta)\xfzero(t)\rho^{1+\lambda}}
 {8g\Tm^\alpha(t)}
 \left[\frac{\Tb^4(t)}{\Ts^4(t)}-1\right].}
 \label{eq:B_bath}
\end{equation}
\end{resultbox}
Using $\xfzero$ rather than the final moment-corrected front keeps the boundary
subsystem sequential; replacing it by $\xf$ would introduce a higher-order
implicit feedback.

Equating \eqref{eq:B_bath} to the HR curvature relation gives one scalar ODE
for the unknown surface radiation temperature:
\begin{resultbox}
\begin{equation}
 \boxed{
 \begin{aligned}
 \dot\Ts(t)
 ={}&\frac{2\Ts^5(t)\Tm^\alpha(t)G(R(t))}
 {\eps(4+\alpha)\,\dfrac{2+\eps}{1-\eps}
 \displaystyle\int_0^t
 \Ts^4(t')\Tm^\alpha(t')G(R(t'))\dif t'}
 \\
 &\times\Bigg\{
 \frac{3(4+\alpha-\beta)\xfzero(t)\rho^{1+\lambda}}
 {8g\Tm^\alpha(t)}
 \left[\frac{\Tb^4(t)}{\Ts^4(t)}-1\right]
 -1+\frac{\eps}{2}\Bigg\}.
 \end{aligned}}
 \label{eq:Ts_bath_ode}
\end{equation}
\end{resultbox}
The bath drive therefore adds only one physical scalar ODE.  It uses $\Tb(t)$
but not $\dot\Tb(t)$, which is advantageous for tabulated data.  In Python,
solve \eqref{eq:Ts_bath_ode} together with
\eqref{eq:surface_material_ode} using \texttt{solve\_ivp} with a stiff method.
The cold onset is formally singular because the base-front history vanishes;
a robust implementation starts at the first positive drive time with a small
positive thermal seed and checks insensitivity to that seed.

After $\Ts(t)$ and $B(t)$ have been determined, the surface- and bath-driven
models are identical.

\section{Monotone material-profile closure}

The material profile is represented by the same radiation shape multiplied by
one monotone ratio factor,
\begin{resultbox}
\begin{equation}
 \boxed{
 \begin{aligned}
 T(x,t)=\Tm(t)
 &\left[
 \left(1-\frac{x}{\xf(t)}\right)
 \left(1+B(t)\frac{x}{\xf(t)}\right)
 \right]^{1/(4+\alpha-\beta)}
 \\
 &\times\exp\left[-s(t)\frac{x}{\xf(t)}\right].
 \end{aligned}}
 \label{eq:material_profile}
\end{equation}
\end{resultbox}
The ratio is
\begin{equation}
 \frac{T(x,t)}{\Tr(x,t)}
 =\frac{\Tm(t)}{\Ts(t)}
 \exp\left[-s(t)\frac{x}{\xf(t)}\right].
 \label{eq:temperature_ratio}
\end{equation}
Positive $s$ narrows the material profile during heating, $s=0$ gives
proportional profiles, and negative $s$ permits retained material energy during
cooling.  This parametrization remains positive for every real $s$ and does
not change the leading front exponent.

\section{Profile moments and their closed forms}

Every spatial integral required by the reduced model can be generated from two
moments,
\begin{equation}
 \boxed{
 M_p(t,z)=\int_0^1
 \left[(1-y)(1+B(t)y)\right]^{p/(4+\alpha-\beta)}
 e^{-zy}\dif y,}
 \label{eq:Mp}
\end{equation}
\begin{equation}
 \boxed{
 N_p(t,z)=\int_0^1
 y\left[(1-y)(1+B(t)y)\right]^{p/(4+\alpha-\beta)}
 e^{-zy}\dif y.}
 \label{eq:Np}
\end{equation}

These integrals have exact forms in terms of Humbert's confluent
hypergeometric function $\Phi_1$:
\begin{align}
 M_p(t,z)
 &=\frac{1}{1+\dfrac{p}{4+\alpha-\beta}}
 \Phi_1\!\left(
 1,-\frac{p}{4+\alpha-\beta};
 2+\frac{p}{4+\alpha-\beta};
 -B(t),-z
 \right),
 \label{eq:Mp_humbert}\\
 N_p(t,z)
 &=\frac{1}
 {\left(1+\dfrac{p}{4+\alpha-\beta}\right)
  \left(2+\dfrac{p}{4+\alpha-\beta}\right)}
 \Phi_1\!\left(
 2,-\frac{p}{4+\alpha-\beta};
 3+\frac{p}{4+\alpha-\beta};
 -B(t),-z
 \right).
 \label{eq:Np_humbert}
\end{align}
The original integral representation of Humbert's function is
\begin{equation}
 \boxed{
 \Phi_1(A,B;C;x,z)
 =\frac{\Gamma(C)}{\Gamma(A)\Gamma(C-A)}
 \int_0^1
 e^{zu}u^{A-1}(1-u)^{C-A-1}
 (1-xu)^{-B}\dif u.}
 \label{eq:humbert_integral}
\end{equation}
The conditions relevant here are $B(t)>-1$ and
$p/(4+\alpha-\beta)>-1$.

SciPy provides \texttt{hyp2f1} for $G(R)$ but does not currently expose
Humbert $\Phi_1$ directly.  Although $\Phi_1$ can be represented by nested
series of one-variable hypergeometric functions, that is not the recommended
production route.  The most transparent and accurate Python evaluation of
\eqref{eq:Mp}--\eqref{eq:Np} is
\texttt{quad}, which uses adaptive QUADPACK integration and
returns an error estimate.  The algebraic endpoint behavior can be supplied
through \texttt{weight="alg"}.  If performance later matters, a fixed
Gauss--Jacobi rule can be validated against \texttt{quad}.

\section{Areal material-energy equation}

The material energy per unit transverse area represented by the reduced
profile is
\begin{resultbox}
\begin{equation}
 \boxed{
 \calU_m(t)
 =\xfzero(t)f\rho^{1-\mu}\Tm^\beta(t)
 M_\beta\!\left(t,\beta s(t)\right).}
 \label{eq:Um_definition}
\end{equation}
\end{resultbox}
The base front $\xfzero$ is used in this closure so that the material-energy
subsystem remains sequential.

Integrating the material equation \eqref{eq:gray_material} over the reduced
profile gives
\begin{resultbox}
\begin{equation}
 \boxed{
 \begin{aligned}
 \dot{\calU}_m(t)
 ={}&\frac{\arad c}{g'}\rho^{1+\lambda'}\xfzero(t)
 \Big[
 \Ts^4(t)\Tm^{-\alpha'}(t)
 M_{4-\alpha'}\!\left(t,-\alpha's(t)\right)
 \\
 &\hspace{30mm}
 -\Tm^{4-\alpha'}(t)
 M_{4-\alpha'}\!\left(t,(4-\alpha')s(t)\right)
 \Big].
 \end{aligned}}
 \label{eq:Um_ode}
\end{equation}
\end{resultbox}
At every ODE right-hand-side evaluation, $s(t)$ is determined from
\begin{resultbox}
\begin{equation}
 \boxed{
 M_\beta\!\left(t,\beta s(t)\right)
 =\frac{\calU_m(t)}
 {\xfzero(t)f\rho^{1-\mu}\Tm^\beta(t)}.}
 \label{eq:s_root}
\end{equation}
\end{resultbox}
The left side is strictly decreasing because
\begin{equation}
 \frac{\partial}{\partial s}
 M_\beta(t,\beta s)
 =-\beta\int_0^1
 y\left[(1-y)(1+B(t)y)\right]^{\beta/(4+\alpha-\beta)}
 e^{-\beta s y}\dif y<0.
 \label{eq:s_monotone}
\end{equation}
Hence the root is unique.  In Python, use the bracketed solver
\texttt{brentq}.  This is a local scalar inversion, not a
Volterra equation; all material-energy history is compressed into the single
ODE state $\calU_m(t)$.

\section{Final front correction}

For any trial material-lag value $s$, define the first total-energy moment
\begin{equation}
 \boxed{
 \begin{aligned}
 \calE_1(t,s)
 ={}&\arad\Ts^4(t)N_4(t,0)
 \\
 &+f\rho^{1-\mu}\Tm^\beta(t)
 N_\beta\!\left(t,\beta s\right).
 \end{aligned}}
 \label{eq:E1}
\end{equation}
Define also the dimensionless transport integrand
\begin{equation}
 \boxed{
 \begin{aligned}
 \calD(t,s)
 ={}&\Ts^4(t)\Tm^\alpha(t)
 \Big\{[1-B(t)]M_\beta\!\left(t,\alpha s\right)
 \\
 &\hspace{29mm}
 +2B(t)N_\beta\!\left(t,\alpha s\right)\Big\}.
 \end{aligned}}
 \label{eq:D_compact}
\end{equation}
The compact form follows from the identity
\begin{equation}
 \begin{aligned}
 &\left[(1-y)(1+B y)\right]^{(\alpha+3)/(4+\alpha-\beta)}
 \left[-\frac{\dif}{\dif y}
 \left((1-y)(1+B y)\right)^{1/(4+\alpha-\beta)}\right]
 \\
 &\qquad=\frac{(1-B)+2By}{4+\alpha-\beta}
 \left[(1-y)(1+B y)\right]^{\beta/(4+\alpha-\beta)}.
 \end{aligned}
 \label{eq:transport_identity}
\end{equation}
All dimensional constants in the physical transport moment are identical in
the numerator and denominator below and therefore cancel.

The final moment-corrected front is
\begin{resultbox}
\begin{equation}
 \boxed{
 \xf^2(t)
 =\xfzero^2(t)
 \frac{\calE_1(t,0)}{\calE_1(t,s(t))}
 \frac{\displaystyle\int_0^t
 \calD\!\left(t',s(t')\right)\dif t'}
 {\displaystyle\int_0^t
 \calD(t',0)\dif t'}.}
 \label{eq:xf_final}
\end{equation}
\end{resultbox}
The denominator is simply the special case $s=0$ of the same integrand.
Numerically the two time integrals are two running scalar quadratures.  Once
$\xf(t)$ is known, the temperature profiles follow algebraically from
\eqref{eq:radiation_profile} and \eqref{eq:material_profile}.

\subsection{Surface-temperature drive}

For prescribed $\Ts(t)$, the evolving quantities are
\begin{itemize}
 \item the physical scalar ODE for $\Tm(t)$;
 \item the physical scalar ODE for $\calU_m(t)$;
 \item the running quadrature defining $\xfzero(t)$;
 \item the two running quadratures in \eqref{eq:xf_final}.
\end{itemize}
At every ODE evaluation:
\begin{enumerate}
 \item interpolate $\Ts(t)$ and evaluate its derivative only for $B(t)$;
 \item evaluate $R(t)$ and $G(R(t))$;
 \item update the base-front quadrature and calculate $\xfzero(t)$;
 \item calculate $B(t)$ from \eqref{eq:B_Ts};
 \item solve the unique scalar equation \eqref{eq:s_root} for $s(t)$;
 \item evaluate the moments and the right-hand sides for $\Tm$, $\calU_m$,
       and the three quadratures;
 \item calculate $\xf(t)$ and the profiles only at requested output times.
\end{enumerate}
Thus the surface-driven model requires
\begin{equation}
 \boxed{
 2\text{ scalar physical ODEs}
 +3\text{ running quadratures}
 +1\text{ scalar root}.}
 \label{eq:complexity_Ts}
\end{equation}

\subsection{Bath-temperature drive}

For prescribed $\Tb(t)$, add only the scalar boundary ODE
\eqref{eq:Ts_bath_ode}.  The remaining calculation is unchanged.  No
derivative of the bath history is needed.  The bath-driven model therefore
requires
\begin{equation}
 \boxed{
 3\text{ scalar physical ODEs}
 +3\text{ running quadratures}
 +1\text{ scalar root}.}
 \label{eq:complexity_Tbath}
\end{equation}
The state dimension is fixed and independent of spatial resolution.

By comparison, a full gray PDE calculation requires hundreds or thousands of
spatial cells, two unknown fields per cell, temperature-dependent face
coefficients, an implicit matrix solve at every timestep, nonlinear opacity
updates, boundary-face treatment, timestep control, and spatial convergence
studies.  The reduced model performs only a small fixed number of scalar
operations per ODE evaluation.

\section{Numerical illustration}

The model was tested for the supplied SiO$_2$ drive using both surface and
bath boundary specifications.  The simulations use a validated $N=1000$
finite-volume gray diffusion solver without a flux limiter.  The bath-driven
simulation uses the Marshak Robin boundary condition directly.

\end{figure}

For the bath-driven gray case, the front error decreases from approximately
$8\%$ at $0.5$ ns to less than $1\%$ near 4 ns.  For the surface-driven gray
case, the front remains near the one-percent level over most of the pulse.
The larger relative errors immediately after the discontinuous turn-on occur
inside the short onset layer where both the HR asymptotics and a numerical
front definition are least reliable.

\section{Accuracy and limitations}

The model combines a first-order HR radiation profile with integrated material
energy and transport moments.  It is not a pointwise solution of the gray PDE
and is not a proof of a uniform $\order(\eps)$ expansion for arbitrary drives.
Its principal approximations are:
\begin{itemize}
 \item one HR-shaped radiation profile;
 \item one scalar material-lag parameter $s(t)$;
 \item use of the base front $\xfzero$ inside the surface-flux and material-energy
       closures;
 \item enforcement of integrated moments rather than the PDEs pointwise.
\end{itemize}
The model is expected to work best for sharp, single-front waves with smooth
power-law material properties.  Strong opacity transitions, multiple fronts,
large excursions with $B(t)\le-1$, or strongly non-HR radiation profiles
indicate that another profile parameter or the full gray PDE may be required.

\section{Self-contained summary of the working model}
\label{sec:summary}

This section can be used independently as an implementation specification.
No profile symbol is introduced; every profile and moment is written
explicitly.

\subsection*{Problem definition and input}

A fixed-density material occupies $x>0$.  Radiation and material temperatures
are $\Tr(x,t)$ and $T(x,t)$.  Use
\begin{equation}
 E_r=\arad\Tr^4,
 \qquad
 U=f\rho^{1-\mu}T^\beta,
 \qquad
 \boxed{\arad=\frac{4\sigma_{\rm SB}}{c}},
 \label{eq:sum_energies}
\end{equation}
\begin{equation}
 \frac{1}{\kappa_R}=gT^\alpha\rho^{-\lambda},
 \qquad
 \frac{1}{\kappa_a}=g'T^{\alpha'}\rho^{-\lambda'}.
 \label{eq:sum_opacities}
\end{equation}
The Rosseland parameters are $(g,\alpha,\lambda)$ and the absorption parameters
are $(g',\alpha',\lambda')$.  Define
\begin{equation}
 \boxed{
 \eps=\frac{\beta}{4+\alpha},
 \qquad
 C=\frac{4\arad c g}{3(4+\alpha)f}
 \rho^{-2+\mu-\lambda}.}
 \label{eq:sum_constants}
\end{equation}
Let
\begin{equation}
 \boxed{\Tm(t)=T(0,t),\qquad \Ts(t)=\Tr(0,t).}
 \label{eq:sum_surface_temperatures}
\end{equation}

For a surface-temperature drive, $\Ts(t)$ is prescribed.  For a bath drive,
$\Tb(t)$ is prescribed and $\Ts(t)$ is determined by the Marshak condition
\begin{equation}
 \boxed{
 F_s(t)=\frac{\arad c}{2}
 [\Tb^4(t)-\Ts^4(t)].}
 \label{eq:sum_marshak}
\end{equation}

\subsection*{Step 1: surface material temperature --- scalar ODE}

For either drive type, solve
\begin{resultbox}
\begin{equation}
 \boxed{
 \frac{\dif}{\dif t}
 [f\rho^{1-\mu}\Tm^\beta(t)]
 =\frac{\arad c}{g'}\rho^{1+\lambda'}
 \Tm^{-\alpha'}(t)[\Ts^4(t)-\Tm^4(t)].}
 \label{eq:sum_Tm_ode}
\end{equation}
\end{resultbox}
Numerically: use \texttt{solve\_ivp(method="Radau")} or
\texttt{solve\_ivp(method="BDF")}.

\subsection*{Step 2: gray storage and base front --- one running quadrature}

Calculate
\begin{equation}
 \boxed{
 R(t)=\frac{\arad\Ts^4(t)}
 {f\rho^{1-\mu}\Tm^\beta(t)}.}
 \label{eq:sum_R}
\end{equation}
The gray storage factor may be evaluated either from its original integral,
\begin{equation}
 \boxed{
 G(R)=(4+\alpha-\beta)
 \int_0^1\frac{\xi^{3+\alpha-\beta}}
 {1+R\xi^{4-\beta}}\dif\xi,}
 \label{eq:sum_G_integral}
\end{equation}
or from the exact closed form
\begin{equation}
 \boxed{
 G(R)={}_2F_1\!\left(
 1,\frac{4+\alpha-\beta}{4-\beta};
 1+\frac{4+\alpha-\beta}{4-\beta};-R\right).}
 \label{eq:sum_G_hyper}
\end{equation}
The defining integral of ${}_2F_1$ is
\begin{equation}
 {}_2F_1(a,b;c;z)
 =\frac{\Gamma(c)}{\Gamma(b)\Gamma(c-b)}
 \int_0^1u^{b-1}(1-u)^{c-b-1}(1-zu)^{-a}\dif u.
 \label{eq:sum_2F1_integral}
\end{equation}
Numerically: use \texttt{hyp2f1}.

The base front is
\begin{resultbox}
\begin{equation}
 \boxed{
 \xfzero^2(t)
 =\frac{2+\eps}{1-\eps}C\Tm^{-\beta}(t)
 \int_0^t\Ts^4(t')\Tm^\alpha(t')G(R(t'))\dif t'.}
 \label{eq:sum_xf0}
\end{equation}
\end{resultbox}
Numerically: store the time integral as one scalar quadrature state.

\subsection*{Step 3: radiation curvature $B(t)$}

For prescribed $\Ts(t)$,
\begin{resultbox}
\begin{equation}
 \boxed{
 \begin{aligned}
 B(t)=\frac{\eps}{2}\Bigg[1
 -\frac{2+\eps}{1-\eps}
 &\frac{\displaystyle\int_0^t
 \Ts^4(t')\Tm^\alpha(t')G(R(t'))\dif t'}
 {\Ts^4(t)\Tm^\alpha(t)G(R(t))}
 \\
 &\times\frac{\dif}{\dif t}\ln\Ts^{4+\alpha}(t)
 \Bigg].
 \end{aligned}}
 \label{eq:sum_B_Ts}
\end{equation}
\end{resultbox}
Numerically: use a \texttt{PchipInterpolator} and its derivative for a
tabulated $\Ts(t)$.

For prescribed $\Tb(t)$, $\Ts(t)$ is an additional scalar ODE,
\begin{resultbox}
\begin{equation}
 \boxed{
 \begin{aligned}
 \dot\Ts(t)
 ={}&\frac{2\Ts^5(t)\Tm^\alpha(t)G(R(t))}
 {\eps(4+\alpha)\,\dfrac{2+\eps}{1-\eps}
 \displaystyle\int_0^t\Ts^4(t')\Tm^\alpha(t')G(R(t'))\dif t'}
 \\
 &\times\Bigg\{
 \frac{3(4+\alpha-\beta)\xfzero(t)\rho^{1+\lambda}}
 {8g\Tm^\alpha(t)}
 \left[\frac{\Tb^4(t)}{\Ts^4(t)}-1\right]
 -1+\frac{\eps}{2}\Bigg\}.
 \end{aligned}}
 \label{eq:sum_Ts_bath_ode}
\end{equation}
\end{resultbox}
and
\begin{equation}
 \boxed{
 B(t)=1-
 \frac{3(4+\alpha-\beta)\xfzero(t)\rho^{1+\lambda}}
 {8g\Tm^\alpha(t)}
 \left[\frac{\Tb^4(t)}{\Ts^4(t)}-1\right].}
 \label{eq:sum_B_bath}
\end{equation}
Numerically: integrate $\Ts(t)$ with the same stiff \texttt{solve\_ivp}
call.  No derivative of $\Tb(t)$ is required.

\subsection*{Step 4: profile moments --- adaptive scalar quadratures}

For any $p$ and $z$, define
\begin{equation}
 \boxed{
 M_p(t,z)=\int_0^1
 [(1-y)(1+B(t)y)]^{p/(4+\alpha-\beta)}e^{-zy}\dif y,}
 \label{eq:sum_M}
\end{equation}
\begin{equation}
 \boxed{
 N_p(t,z)=\int_0^1
 y[(1-y)(1+B(t)y)]^{p/(4+\alpha-\beta)}e^{-zy}\dif y.}
 \label{eq:sum_N}
\end{equation}
Their exact Humbert forms are
\begin{align}
 M_p(t,z)
 &=\frac{1}{1+\dfrac{p}{4+\alpha-\beta}}
 \Phi_1\!\left(
 1,-\frac{p}{4+\alpha-\beta};
 2+\frac{p}{4+\alpha-\beta};-B(t),-z\right),
 \label{eq:sum_M_humbert}\\
 N_p(t,z)
 &=\frac{1}
 {\left(1+\dfrac{p}{4+\alpha-\beta}\right)
  \left(2+\dfrac{p}{4+\alpha-\beta}\right)}
 \Phi_1\!\left(
 2,-\frac{p}{4+\alpha-\beta};
 3+\frac{p}{4+\alpha-\beta};-B(t),-z\right),
 \label{eq:sum_N_humbert}
\end{align}
with
\begin{equation}
 \boxed{
 \Phi_1(A,B;C;x,z)
 =\frac{\Gamma(C)}{\Gamma(A)\Gamma(C-A)}
 \int_0^1e^{zu}u^{A-1}(1-u)^{C-A-1}
 (1-xu)^{-B}\dif u.}
 \label{eq:sum_humbert_integral}
\end{equation}
Numerically: SciPy has no direct $\Phi_1$ function, so use
\texttt{quad} for $M_p$ and $N_p$.  The returned error estimate
should be checked during validation.

\subsection*{Step 5: material lag $s(t)$ --- one scalar ODE and one root}

Evolve the areal material energy
\begin{resultbox}
\begin{equation}
 \boxed{
 \begin{aligned}
 \dot{\calU}_m(t)
 ={}&\frac{\arad c}{g'}\rho^{1+\lambda'}\xfzero(t)
 \Big[
 \Ts^4(t)\Tm^{-\alpha'}(t)
 M_{4-\alpha'}(t,-\alpha's(t))
 \\
 &\hspace{27mm}
 -\Tm^{4-\alpha'}(t)
 M_{4-\alpha'}(t,(4-\alpha')s(t))
 \Big].
 \end{aligned}}
 \label{eq:sum_U_ode}
\end{equation}
\end{resultbox}
At each ODE evaluation, solve
\begin{resultbox}
\begin{equation}
 \boxed{
 M_\beta(t,\beta s(t))
 =\frac{\calU_m(t)}
 {\xfzero(t)f\rho^{1-\mu}\Tm^\beta(t)}.}
 \label{eq:sum_s_root}
\end{equation}
\end{resultbox}
Numerically: use \texttt{brentq}; the root is unique because the
left side decreases strictly with $s$.

\subsection*{Step 6: final front --- two running quadratures}

For a trial value $s$, define
\begin{equation}
 \boxed{
 \calE_1(t,s)=\arad\Ts^4(t)N_4(t,0)
 +f\rho^{1-\mu}\Tm^\beta(t)N_\beta(t,\beta s),}
 \label{eq:sum_E1}
\end{equation}
\begin{equation}
 \boxed{
 \begin{aligned}
 \calD(t,s)=\Ts^4(t)\Tm^\alpha(t)
 \Big\{[1-B(t)]M_\beta(t,\alpha s)
 +2B(t)N_\beta(t,\alpha s)\Big\}.
 \end{aligned}}
 \label{eq:sum_D}
\end{equation}
Then
\begin{resultbox}
\begin{equation}
 \boxed{
 \xf^2(t)=\xfzero^2(t)
 \frac{\calE_1(t,0)}{\calE_1(t,s(t))}
 \frac{\displaystyle\int_0^t\calD(t',s(t'))\dif t'}
 {\displaystyle\int_0^t\calD(t',0)\dif t'}.}
 \label{eq:sum_xf}
\end{equation}
\end{resultbox}
Numerically: advance the numerator and denominator as two scalar running
quadratures.

\subsection*{Step 7: final profiles --- algebraic evaluation}

For $0\le x\le\xf(t)$,
\begin{resultbox}
\begin{equation}
 \boxed{
 \Tr(x,t)=\Ts(t)
 \left[
 \left(1-\frac{x}{\xf(t)}\right)
 \left(1+B(t)\frac{x}{\xf(t)}\right)
 \right]^{1/(4+\alpha-\beta)},}
 \label{eq:sum_Tr_profile}
\end{equation}
\begin{equation}
 \boxed{
 \begin{aligned}
 T(x,t)=\Tm(t)
 &\left[
 \left(1-\frac{x}{\xf(t)}\right)
 \left(1+B(t)\frac{x}{\xf(t)}\right)
 \right]^{1/(4+\alpha-\beta)}
 \\
 &\times\exp\left[-s(t)\frac{x}{\xf(t)}\right].
 \end{aligned}}
 \label{eq:sum_T_profile}
\end{equation}
\end{resultbox}
Set both temperatures to zero beyond the sharp front.

\subsection*{Complete numerical workflow and cost}

For prescribed $\Ts(t)$:
\begin{enumerate}
 \item integrate the two physical scalar ODEs for $\Tm(t)$ and $\calU_m(t)$;
 \item advance the base-front quadrature and the two final-front quadratures;
 \item evaluate $G$ with \texttt{hyp2f1};
 \item evaluate $M_p,N_p$ with \texttt{quad};
 \item solve the unique $s(t)$ root with \texttt{brentq};
 \item evaluate $\xf$, $\Tr$, and $T$ algebraically at output times.
\end{enumerate}
The cost is
\begin{equation}
 \boxed{
 2\text{ scalar physical ODEs}
 +3\text{ scalar quadratures}
 +1\text{ scalar root}.}
 \label{eq:sum_cost_Ts}
\end{equation}

For prescribed $\Tb(t)$, add only the scalar ODE for $\Ts(t)$; all later steps
are identical.  The cost is
\begin{equation}
 \boxed{
 3\text{ scalar physical ODEs}
 +3\text{ scalar quadratures}
 +1\text{ scalar root}.}
 \label{eq:sum_cost_Tbath}
\end{equation}
No spatial grid, diffusion matrix, PDE iteration, or time-dependent integral
equation appears in either reduced model.
